package com.example.demo;

import java.util.List;
import jakarta.persistence.*;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.web.bind.annotation.*;

@Entity
class Student {
    @Id @GeneratedValue
    public Integer id;
    public String name, dept;
}

interface StudentRepository extends JpaRepository<Student, Integer> { }

@SpringBootApplication
@RestController
public class DemoApplication {

    private final StudentRepository repo;

    public DemoApplication(StudentRepository repo) { this.repo = repo; }

    public static void main(String[] args) {
        SpringApplication.run(DemoApplication.class, args);
    }

    @GetMapping("/add/{name}/{dept}")
    public Student add(@PathVariable String name, @PathVariable String dept) {
        Student s = new Student();
        s.name = name;
        s.dept = dept;
        return repo.save(s);
    }

    @GetMapping("/sort")
    public List<Student> sort() {
        return repo.findAll(Sort.by("name"));
    }

    @GetMapping("/page/{page}/{size}")
    public List<Student> page(@PathVariable int page, @PathVariable int size) {
        return repo.findAll(PageRequest.of(page, size, Sort.by("name"))).getContent();
    }
}
