package com.example.demo;

import java.util.List;
import jakarta.persistence.*;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.web.bind.annotation.*;

@Entity
class Train {
    @Id @GeneratedValue
    public Integer id;
    public String name, source, destination;
}

interface TrainRepository extends JpaRepository<Train, Integer> {
    List<Train> findBySource(String source);
}

@SpringBootApplication
@RestController
public class DemoApplication {

    private final TrainRepository repo;

    public DemoApplication(TrainRepository repo) { this.repo = repo; }

    public static void main(String[] args) {
        SpringApplication.run(DemoApplication.class, args);
    }

    @PostMapping("/train")
    public Train add(@RequestBody Train t) { return repo.save(t); }

    @GetMapping("/train")
    public List<Train> all() { return repo.findAll(); }

    @GetMapping("/train/{id}")
    public Train get(@PathVariable int id) { return repo.findById(id).orElse(null); }

    @GetMapping("/train/source/{source}")
    public List<Train> bySource(@PathVariable String source) { return repo.findBySource(source); }

    @PutMapping("/train")
    public Train update(@RequestBody Train t) { return repo.save(t); }

    @DeleteMapping("/train/{id}")
    public String delete(@PathVariable int id) {
        repo.deleteById(id);
        return "Deleted";
    }
}
