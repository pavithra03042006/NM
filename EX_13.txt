package com.example.demo;

import java.util.List;
import jakarta.persistence.*;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.web.bind.annotation.*;

@Entity
class Ticket {
    @Id @GeneratedValue
    public Integer id;
    public String source, destination;
}

interface TicketRepository extends JpaRepository<Ticket, Integer> {
    List<Ticket> findBySource(String source);
}

@SpringBootApplication
@RestController
public class DemoApplication {

    private final TicketRepository repo;

    public DemoApplication(TicketRepository repo) { this.repo = repo; }

    public static void main(String[] args) {
        SpringApplication.run(DemoApplication.class, args);
    }

    @PostMapping("/ticket")
    public Ticket add(@RequestBody Ticket t) { return repo.save(t); }

    @GetMapping("/ticket")
    public List<Ticket> all() { return repo.findAll(); }

    @GetMapping("/ticket/{id}")
    public Ticket get(@PathVariable int id) { return repo.findById(id).orElse(null); }

    @GetMapping("/ticket/source/{source}")
    public List<Ticket> bySource(@PathVariable String source) { return repo.findBySource(source); }

    @PutMapping("/ticket")
    public Ticket update(@RequestBody Ticket t) { return repo.save(t); }

    @DeleteMapping("/ticket/{id}")
    public String delete(@PathVariable int id) {
        repo.deleteById(id);
        return "Ticket Cancelled";
    }
}
